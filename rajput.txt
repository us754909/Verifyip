<div class="container">
    <h2>Verify Your Identity 🛡️</h2>
    <p id="ip-text">Fetching your secure IP...</p>
    <button class="btn" id="verify-btn" style="display: none;" onclick="sendIP()">Verify Now</button>
</div>

<script>
    let tg = window.Telegram.WebApp;
    tg.expand(); // Full screen open hoga

    let userIP = "";

    // IP nikalne ke liye API call
    fetch('https://api.ipify.org?format=json')
        .then(response => response.json())
        .then(data => {
            userIP = data.ip;
            document.getElementById('ip-text').innerText = "Your IP: " + userIP;
            document.getElementById('verify-btn').style.display = "inline-block";
        })
        .catch(err => {
            document.getElementById('ip-text').innerText = "Error fetching IP. Try again.";
        });

    function sendIP() {
        // Ye line BotBusiness ko IP data bhejegi
        tg.sendData(userIP);
        tg.close();
    }
</script>
